# Configuración de Sesión y Timeouts
# Copiar este archivo a .env.local y ajustar los valores según necesidad

# Duración máxima de la sesión (en milisegundos)
# Default: 24 horas = 86400000 ms
NEXT_PUBLIC_SESSION_DURATION_MS=86400000

# Tiempo de inactividad antes de cerrar sesión (en milisegundos)
# Default: 30 minutos = 1800000 ms
NEXT_PUBLIC_INACTIVITY_TIMEOUT_MS=1800000

# Intervalo de verificación de expiración (en milisegundos)
# Default: 1 minuto = 60000 ms
NEXT_PUBLIC_CHECK_INTERVAL_MS=60000

# Tiempo antes de mostrar advertencia de inactividad (en milisegundos)
# Default: 2 minutos = 120000 ms
NEXT_PUBLIC_WARNING_BEFORE_TIMEOUT_MS=120000

# Ejemplos de configuraciones comunes:

# Desarrollo (timeouts cortos para testing)
# NEXT_PUBLIC_SESSION_DURATION_MS=3600000        # 1 hora
# NEXT_PUBLIC_INACTIVITY_TIMEOUT_MS=300000       # 5 minutos
# NEXT_PUBLIC_WARNING_BEFORE_TIMEOUT_MS=60000    # 1 minuto

# Producción (timeouts largos)
# NEXT_PUBLIC_SESSION_DURATION_MS=86400000       # 24 horas
# NEXT_PUBLIC_INACTIVITY_TIMEOUT_MS=3600000      # 1 hora
# NEXT_PUBLIC_WARNING_BEFORE_TIMEOUT_MS=300000   # 5 minutos

# NEXT_PUBLIC_INACTIVITY_TIMEOUT_MS=600000       # 10 minutos
# NEXT_PUBLIC_WARNING_BEFORE_TIMEOUT_MS=120000   # 2 minutos
